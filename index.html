🔧 Firebase 설정 방법
  1. Firebase 콘솔에서 새 프로젝트를 만드세요.
  2. 프로젝트 설정(⚙️) → 내 앱 → 웹 앱 추가 → firebaseConfig 값을 복사하세요.
  3. firebase-config.js의 YOUR_* 부분을 실제 값으로 교체하세요.
  4. Firestore Database를 생성하고 아래 보안 규칙을 적용하세요.
  5. 페이지를 새로고침하면 자동으로 연결됩니다.
Firestore 보안 규칙 — 테스트용 (개발 중에만 사용, 실서비스 절대 금지) 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /{document=**} {
      allow read, write: if true;
    }
  }
}
로그인 사용자만 허용 (실서비스 권장) 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /events/{eventId} {
      allow read, write: if request.auth != null;
      match /participants/{id} { allow read, write: if request.auth != null; }
      match /sponsors/{id}     { allow read, write: if request.auth != null; }
      match /sponsorItems/{id} { allow read, write: if request.auth != null; }
      match /staff/{id}        { allow read, write: if request.auth != null; }
    }
  }
}
